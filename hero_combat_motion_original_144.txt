func play_tank_skill_fx(kind: int) -> void:
 if tank_fx_root == null:
  tank_fx_root = Node3D.new()
  tank_fx_root.name = "TankSkillVisual"
  add_child(tank_fx_root)
 tank_fx_kind = kind
 tank_fx_time = 1.25
 for child in tank_fx_root.get_children():
  child.queue_free()
 if kind == 8: # Taunt: rotating gold challenge ring.
  for i in range(8):
   var angle: float = float(i) * TAU / 8.0
   VisualFactory.box(tank_fx_root, Vector3(0.13, 0.08, 0.13), Vector3(cos(angle) * 0.78, 1.65, sin(angle) * 0.78), Color(1.0, 0.68, 0.12))
 elif kind == 9: # Interception: wide blue protective ring.
  for i in range(12):
   var angle: float = float(i) * TAU / 12.0
   VisualFactory.box(tank_fx_root, Vector3(0.17, 0.10, 0.17), Vector3(cos(angle) * 1.12, 1.1, sin(angle) * 1.12), Color(0.24, 0.70, 1.0))
 tank_fx_root.visible = true

# Refreshes aura for the duration of the actual combat status, without spawning duplicates.
func set_tank_defense_state(kind: int) -> void:
 if kind < 0:
  tank_fx_time = 0.0
  if tank_fx_root != null:
   tank_fx_root.visible = false
  tank_fx_kind = -1
  return
 if tank_fx_root == null or tank_fx_kind != kind:
  play_tank_skill_fx(kind)
 tank_fx_time = 1.5
 if tank_fx_root != null:
  tank_fx_root.visible = true

func play_attack() -> void:
 attack_time = strike_duration / class_attack_tempo

func play_hit() -> void:
 hit_time = 0.22
 hit_recoil = 1.0
 hit_brace = 1.0
 hit_counter += 1
 hit_side = -1.0 if hit_counter % 2 == 0 else 1.0
 # Visual hit response is class-dependent; does not change HP or combat timers.
 hit_time = 0.30 if class_id in [0, 1] else 0.22

# Elemental flinches have different silhouettes and automatically decay.
func play_elemental_hit(element: int) -> void:
 play_hit()
 elemental_hit_id = clampi(element, 0, 9)
 elemental_hit_time = 0.48 if elemental_hit_id == 7 else 0.32

func _process(delta: float) -> void:
 clock += delta
 if tank_fx_time > 0.0:
  tank_fx_time = maxf(0.0, tank_fx_time - delta)
  if tank_fx_root != null:
   tank_fx_root.rotation.y += delta * (3.4 if tank_fx_kind == 8 else -1.9)
   tank_fx_root.scale = Vector3.ONE * (0.85 + 0.15 * sin(clock * 9.0))
   tank_fx_root.visible = tank_fx_time > 0.0
 var walking: bool = false
 var team: Node = get_parent().get_parent()
 if team != null:
  walking = bool(team.get("is_walking"))
 hit_time = maxf(0.0, hit_time - delta)
 elemental_hit_time = maxf(0.0, elemental_hit_time - delta)
 var elemental_strength: float = elemental_hit_time / (0.48 if elemental_hit_id == 7 else 0.32) if elemental_hit_time > 0.0 else 0.0
 hit_recoil = move_toward(hit_recoil, 0.0, delta * 5.0)
 hit_brace = move_toward(hit_brace, 0.0, delta * (3.0 if class_id in [0, 1] else 4.5))
 motion_blend = move_toward(motion_blend, 1.0 if walking else 0.0, delta * 4.0)
 rest_breath = sin(clock * 2.1) * 0.018
 position.y = rest_breath + absf(sin(clock * 9.0 * class_rhythm)) * 0.055 * motion_blend * class_bob
 # Tank braces, DD leans into movement, healer remains more upright.
 var team_role: int = 1
 if team != null:
  var slot: int = int(str(get_parent().name).trim_prefix("Hero")) - 1
  var roles: Variant = team.get("role_ids")
  if roles is Array and slot >= 0 and slot < roles.size():
   team_role = int(roles[slot])
 gait_phase += delta * lerpf(1.6, 8.6, motion_blend) * class_rhythm
 lean_amount = move_toward(lean_amount, (0.075 if team_role == 1 else 0.035) * motion_blend, delta * 0.4)
 rotation.x = lean_amount + sin(gait_phase * 2.0) * 0.012 * motion_blend + attack_body_pitch - hit_recoil * 0.075
 # Lateral center-of-mass transfer follows alternating planted feet.
 # Smoothed so stopping and starting do not cause visible snapping.
 var lateral_target: float = sin(gait_phase) * 0.042 * motion_blend * class_stride
 if team_role == 0:
  lateral_target *= 0.65
 elif team_role == 2:
  lateral_target *= 0.8
 weight_shift = move_toward(weight_shift, lateral_target, delta * 0.45)
 # Impact weight transfers to the supporting foot, then returns to neutral.
 var brace_strength: float = 0.032 if class_id in [0, 1] else 0.060
 position.x = weight_shift + hit_side * hit_brace * brace_strength
 stride_roll = move_toward(stride_roll, -sin(gait_phase) * 0.055 * motion_blend, delta * 0.6)
 torso_turn = move_toward(torso_turn, sin(gait_phase) * 0.075 * motion_blend, delta * 0.7)
 var stride: float = lerpf(0.018, 0.53, motion_blend) * class_stride
 if team_role == 0: stride *= 0.7
 if team_role == 2: stride *= 0.8
 # Smooth both hip angles so the legs settle instead of snapping to idle.
 var hip_target: float = sin(gait_phase) * stride
 left_leg.rotation.x = lerpf(left_leg.rotation.x, hip_target, minf(1.0, delta * 13.0))
 right_leg.rotation.x = lerpf(right_leg.rotation.x, -hip_target, minf(1.0, delta * 13.0))
 left_leg.rotation.z = 0.035 * motion_blend
 right_leg.rotation.z = -0.035 * motion_blend
 # Swing leg bends during recovery; planted leg stays comparatively straight.
 var left_phase: float = sin(gait_phase)
 var right_phase: float = -left_phase
 var left_bend: float = maxf(0.0, -left_phase) * 0.72 * motion_blend
 var right_bend: float = maxf(0.0, -right_phase) * 0.72 * motion_blend
 # A lifted foot rises on the return step; the planted foot remains lower.
 foot_plant_left = lerpf(foot_plant_left, maxf(0.0, -left_phase) * motion_blend, minf(1.0, delta * 11.0))
 foot_plant_right = lerpf(foot_plant_right, maxf(0.0, -right_phase) * motion_blend, minf(1.0, delta * 11.0))
 left_leg.position.y = foot_plant_left * 0.045
 right_leg.position.y = foot_plant_right * 0.045
 # Knees briefly absorb the force of a hit without changing game physics.
 var knee_brace: float = hit_brace * (0.10 if class_id in [0, 1] else 0.16)
 left_knee.rotation.x = lerpf(left_knee.rotation.x, left_bend + knee_brace, minf(1.0, delta * 12.0))
 right_knee.rotation.x = lerpf(right_knee.rotation.x, right_bend + knee_brace, minf(1.0, delta * 12.0))
 # Ankles compensate for knee rotation and roll the toe forward on landing.
 left_foot.rotation.x = -left_knee.rotation.x * 0.60 - maxf(0.0, left_phase) * 0.18 * motion_blend
 right_foot.rotation.x = -right_knee.rotation.x * 0.60 - maxf(0.0, right_phase) * 0.18 * motion_blend
 # Brief stagger on receiving damage, no permanent transform changes.
 rotation.z = stride_roll + attack_body_roll + hit_side * hit_brace * 0.045 + (sin(hit_time * 27.0) * (hit_time / (0.30 if class_id in [0, 1] else 0.22)) * (0.10 if class_id in [0, 1] else 0.16) if hit_time > 0.0 else 0.0)
 # Tank stance remains tied to active shield/taunt status, not the short cast animation.
 var desired_tank_pose: int = tank_fx_kind if tank_fx_root != null and tank_fx_root.visible and team_role == 0 else -1
 if desired_tank_pose != tank_pose_kind:
  tank_pose_blend = move_toward(tank_pose_blend, 0.0, delta * 5.0)
  if tank_pose_blend <= 0.01:
   tank_pose_kind = desired_tank_pose
 else:
  tank_pose_blend = move_toward(tank_pose_blend, 1.0 if desired_tank_pose >= 0 else 0.0, delta * 5.0)
 var swing: float = lerpf(0.045, 0.34, motion_blend)
 if attack_time > 0.0:
  attack_time = maxf(0.0, attack_time - delta)
  var progress: float = 1.0 - attack_time / (strike_duration / class_attack_tempo)
  var strike: float = sin(progress * PI)
  # Wind-up, forward drive and recovery; role/class shapes the pose.
  var anticipation: float = sin(clampf(progress / 0.32, 0.0, 1.0) * PI) * (1.0 - progress)
  var drive: float = sin(clampf((progress - 0.18) / 0.82, 0.0, 1.0) * PI)
  var melee: bool = class_id in [0, 1, 5, 10]
  var force: float = 1.0 if class_id in [0, 1] else (0.72 if melee else 0.38)
  attack_body_pitch = lerpf(attack_body_pitch, (-0.12 * anticipation + 0.16 * drive) * force, minf(1.0, delta * 14.0))
  attack_body_yaw = lerpf(attack_body_yaw, (drive - anticipation * 0.4) * (0.16 if melee else 0.055), minf(1.0, delta * 13.0))
  attack_body_roll = lerpf(attack_body_roll, sin(progress * PI) * (0.045 if melee else 0.018), minf(1.0, delta * 12.0))
  rotation.y = torso_turn + attack_body_yaw
  rotation.x = lean_amount + attack_body_pitch - hit_recoil * 0.075
  rotation.z = stride_roll + attack_body_roll + hit_side * hit_brace * 0.045 + (sin(hit_time * 27.0) * (hit_time / (0.30 if class_id in [0, 1] else 0.22)) * (0.10 if class_id in [0, 1] else 0.16) if hit_time > 0.0 else 0.0)
  # Independent elbows: hands lead the gesture rather than rigid arms swinging.
  var bend: float = sin(progress * PI) * (0.78 if class_id in [4, 5, 10] else 0.48)
  left_forearm.rotation.x = lerpf(left_forearm.rotation.x, -bend, minf(1.0, delta * 13.0))
  right_forearm.rotation.x = lerpf(right_forearm.rotation.x, -bend * 1.25, minf(1.0, delta * 13.0))
  match class_id:
   0: # Krieger: two-handed overhead chop
    left_arm.rotation.x = -strike * 0.85
    right_arm.rotation.x = -strike * 1.85
    right_arm.rotation.z = -strike * 0.35
    rotation.y = torso_turn + attack_body_yaw + sin(progress * PI) * 0.10
   1: # Paladin: shield brace and radiant hammer
    left_arm.rotation.x = -0.65 * strike
    left_arm.rotation.z = -0.35 * strike
    right_arm.rotation.x = -1.55 * strike
    right_arm.rotation.z = 0.20 * strike
   2: # Druide: symmetrical, sweeping vine gesture
    left_arm.rotation.x = -1.20 * strike
    right_arm.rotation.x = -0.95 * strike
    left_arm.rotation.z = -0.40 * strike
    right_arm.rotation.z = 0.40 * strike
   3: # Schamane: staff raise and totem beat
    left_arm.rotation.x = -0.40 * strike
    right_arm.rotation.x = -1.55 * strike
    right_arm.rotation.y = sin(progress * TAU) * 0.22
   4: # Jaeger: draw and release
    left_arm.rotation.x = -0.90 * strike
    right_arm.rotation.x = 0.82 * strike
    right_arm.rotation.y = -0.52 * strike
   5: # Schurke: alternating rapid strikes
    left_arm.rotation.x = -absf(sin(progress * TAU)) * 1.5
    right_arm.rotation.x = -absf(cos(progress * TAU)) * 1.5
    rotation.y = torso_turn + attack_body_yaw + sin(progress * TAU) * 0.13
   6: # Magier: wide arcane cast
    left_arm.rotation.x = -1.10 * strike
    right_arm.rotation.x = -1.40 * strike
    right_arm.rotation.z = 0.52 * strike
   7: # Hexer: asymmetric shadow grasp
    left_arm.rotation.x = -0.45 * strike
    left_arm.rotation.z = -0.45 * strike
    right_arm.rotation.x = -1.62 * strike
    right_arm.rotation.y = 0.50 * strike
   8: # Beschwoerer: open both arms to invoke
    left_arm.rotation.x = -0.72 * strike
    right_arm.rotation.x = -0.72 * strike
    left_arm.rotation.z = -0.72 * strike
    right_arm.rotation.z = 0.72 * strike
   9: # Priester: raised hands for blessing or shadow cast
    left_arm.rotation.x = -1.35 * strike
    right_arm.rotation.x = -1.35 * strike
    left_arm.rotation.z = -0.22 * strike
    right_arm.rotation.z = 0.22 * strike
   10: # Raufbold: alternating close-range punches
    left_arm.rotation.x = -absf(sin(progress * TAU)) * 1.35
    right_arm.rotation.x = -absf(sin(progress * TAU + PI * 0.5)) * 1.65
    rotation.y = torso_turn + attack_body_yaw + sin(progress * TAU) * 0.16
  halo.visible = class_id in [1, 2, 3, 6, 7, 8, 9]
